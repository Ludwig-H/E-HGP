// Lambda(A,B,C) : credit diametral de NOEUD, pas de point.
//
//   Lambda = somme_i min sur les huit triples d'extremites de
//            (C_i^g - A_i^a)(B_i^b - C_i^g)
//          = min sur a dans A, b dans B, z dans C de (z-a).(b-z).
//
// Pour a_i,b_i fixes la fonction de z_i est une parabole CONCAVE : son minimum
// sur l'intervalle est a une extremite. Vingt-quatre produits, deux additions.
//
// Lambda > 0 credite TOUS les PointId du noeud C d'un coup. Le cout devient
// donc un nombre de NOEUDS visites, pas un nombre de paires.
#include <cstdio>
#include <algorithm>
#include <array>
#include <vector>
using i64 = long long;

struct Box { i64 lo[3], hi[3]; };

i64 Lambda(const Box& A, const Box& B, const Box& C) {
  i64 s = 0;
  for (int i = 0; i < 3; ++i) {
    i64 m = 0; bool first = true;
    for (int g = 0; g < 2; ++g) {
      const i64 z = g ? C.hi[i] : C.lo[i];
      for (int a = 0; a < 2; ++a)
        for (int b = 0; b < 2; ++b) {
          const i64 v = (z - (a ? A.hi[i] : A.lo[i])) * ((b ? B.hi[i] : B.lo[i]) - z);
          if (first || v < m) { m = v; first = false; }
        }
    }
    s += m;
  }
  return s;
}

struct Node { Box box; int begin, end, left, right; };

int build(std::vector<Node>& nd, std::vector<std::array<i64,3>>& pts, int b, int e, int leaf) {
  Node n{}; n.begin=b; n.end=e; n.left=n.right=-1;
  for (int i=0;i<3;++i){ n.box.lo[i]=pts[b][i]; n.box.hi[i]=pts[b][i]; }
  for (int t=b;t<e;++t) for(int i=0;i<3;++i){
    n.box.lo[i]=std::min(n.box.lo[i],pts[t][i]); n.box.hi[i]=std::max(n.box.hi[i],pts[t][i]); }
  const int id=(int)nd.size(); nd.push_back(n);
  if (e-b>leaf){
    int ax=0; i64 best=-1;
    for(int i=0;i<3;++i){ const i64 w=nd[id].box.hi[i]-nd[id].box.lo[i]; if(w>best){best=w;ax=i;} }
    const int mid=(b+e)/2;
    std::nth_element(pts.begin()+b,pts.begin()+mid,pts.begin()+e,
      [ax](const std::array<i64,3>&x,const std::array<i64,3>&y){return x[ax]<y[ax];});
    const int l=build(nd,pts,b,mid,leaf); const int r=build(nd,pts,mid,e,leaf);
    nd[id].left=l; nd[id].right=r;
  }
  return id;
}

int main(){
  const int m=25000;
  std::vector<std::array<i64,3>> A(m);
  for(int i=0;i<m;++i) A[i]={0,(i64)(i/200),(i64)(i%200)};
  std::vector<Node> tree; tree.reserve(2*m); build(tree,A,0,m,8);
  Box FB{{60000,10000,1000},{60000,10124,1199}};

  for (int k : {1,4,16}) {
    long long evals=0, credited_nodes=0, closed=0, rects=0, mass_closed=0, mass_tot=0;
    for (i64 y0=0;y0<=124;y0+=k){
      const i64 y1=std::min<i64>(124,y0+k-1);
      Box SA{{0,y0,0},{0,y1,199}};
      const long long mass=(y1-y0+1)*200LL*(long long)m; mass_tot+=mass; ++rects;
      // DFS sur l'arbre des temoins ; un noeud credite n'est pas descendu.
      long long cred=0; std::vector<int> st{0};
      while(!st.empty() && cred<10){
        const int id=st.back(); st.pop_back(); const Node& n=tree[id];
        // plage disjointe de la tranche ancre : les y de la tranche sont exclus
        if (n.box.lo[1]>=y0 && n.box.hi[1]<=y1) continue;
        ++evals;
        if (Lambda(SA,FB,n.box)>0){ cred += n.end-n.begin; ++credited_nodes; continue; }
        if (n.left<0) continue;
        st.push_back(n.left); st.push_back(n.right);
      }
      if (cred>=10){ ++closed; mass_closed+=mass; }
    }
    printf("tranches de %2d : %3lld rectangles | fermes %lld | masse fermee %.2f%% | "
           "evaluations Lambda %lld | noeuds credites %lld\n",
           k,rects,closed,100.0*mass_closed/mass_tot,evals,credited_nodes);
  }
  return 0;
}
