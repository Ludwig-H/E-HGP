// WSPD (Callahan-Kosaraju) : le front factorise a-t-il O(n) rectangles ?
// Puis : chaque rectangle bien separe se ferme-t-il par credit de NOEUD ?
#include <cstdio>
#include <cstdlib>
#include <cmath>
#include <algorithm>
#include <array>
#include <vector>
#include <string>
using i64 = long long; using i128 = __int128;
struct Box { i64 lo[3], hi[3]; };
static i64 Lambda(const Box&A,const Box&B,const Box&C){
  i64 s=0;
  for(int i=0;i<3;++i){ i64 m=0; bool f=true;
    for(int g=0;g<2;++g){ const i64 z=g?C.hi[i]:C.lo[i];
      for(int a=0;a<2;++a) for(int b=0;b<2;++b){
        const i64 v=(z-(a?A.hi[i]:A.lo[i]))*((b?B.hi[i]:B.lo[i])-z);
        if(f||v<m){m=v;f=false;} } }
    s+=m; }
  return s;
}
static i64 maxsq(const Box&P,const Box&Q){
  i64 s=0;
  for(int i=0;i<3;++i){ i64 m=0;
    for(int p=0;p<2;++p) for(int q=0;q<2;++q){
      const i64 d=(p?P.hi[i]:P.lo[i])-(q?Q.hi[i]:Q.lo[i]); m=std::max(m,d*d); }
    s+=m; }
  return s;
}
struct Node{ Box box; int begin,end,left,right; double c[3],rad; };
static int build(std::vector<Node>&nd,std::vector<std::array<i64,3>>&p,int b,int e,int leaf){
  Node n{}; n.begin=b;n.end=e;n.left=n.right=-1;
  for(int i=0;i<3;++i){n.box.lo[i]=p[b][i];n.box.hi[i]=p[b][i];}
  for(int t=b;t<e;++t)for(int i=0;i<3;++i){n.box.lo[i]=std::min(n.box.lo[i],p[t][i]);n.box.hi[i]=std::max(n.box.hi[i],p[t][i]);}
  double r2=0; for(int i=0;i<3;++i){ n.c[i]=0.5*(double)(n.box.lo[i]+n.box.hi[i]);
    const double h=0.5*(double)(n.box.hi[i]-n.box.lo[i]); r2+=h*h; }
  n.rad=std::sqrt(r2);
  const int id=(int)nd.size(); nd.push_back(n);
  if(e-b>leaf){ int ax=0;i64 best=-1;
    for(int i=0;i<3;++i){const i64 w=nd[id].box.hi[i]-nd[id].box.lo[i]; if(w>best){best=w;ax=i;}}
    // split geometrique (fair split) : au milieu de l'axe le plus long
    const i64 mid=(nd[id].box.lo[ax]+nd[id].box.hi[ax])/2;
    auto it=std::partition(p.begin()+b,p.begin()+e,[ax,mid](const std::array<i64,3>&x){return x[ax]<=mid;});
    int m=(int)(it-p.begin());
    if(m==b||m==e){ m=(b+e)/2; std::nth_element(p.begin()+b,p.begin()+m,p.begin()+e,
        [ax](const std::array<i64,3>&x,const std::array<i64,3>&y){return x[ax]<y[ax];}); }
    nd[id].left=build(nd,p,b,m,leaf); nd[id].right=build(nd,p,m,e,leaf); }
  return id;
}
int main(int argc,char**argv){
  int n = argc>1?atoi(argv[1]):4000;
  const int coord = argc>2?atoi(argv[2]):65535;
  const double s = argc>3?atof(argv[3]):4.0;
  unsigned long long r=88172645463325252ULL;
  auto rnd=[&](i64 hi){ r^=r<<13;r^=r>>7;r^=r<<17; return (i64)(r%(unsigned long long)hi); };
  std::vector<std::array<i64,3>> P(n);
  const char*fam=getenv("FAM");
  if(fam&&std::string(fam)=="deuxplans"){
    const int m=n/2; P.resize(2*m);
    for(int i=0;i<m;++i){ const i64 u=i/200, v=i%200;
      P[i]={0,u,v}; P[m+i]={60000,10000+u,1000+v}; }
    n=(int)P.size();
  } else for(int i=0;i<n;++i) P[i]={rnd(coord),rnd(coord),rnd(coord)};
  std::vector<Node> T; T.reserve(8*n); build(T,P,0,n,1);
  auto sep=[&](int ia,int ib){ const Node&A=T[ia];const Node&B=T[ib];
    double d2=0; for(int i=0;i<3;++i){const double t=A.c[i]-B.c[i]; d2+=t*t;}
    return std::sqrt(d2)-A.rad-B.rad >= s*std::max(A.rad,B.rad); };
  const int need[3]={10,9,8};
  long long pairs=0, mass_tot=0, closed[3]={0,0,0}, evals=0, cnodes=0;
  std::vector<std::pair<int,int>> st{{0,0}};
  while(!st.empty()){
    auto[ia,ib]=st.back(); st.pop_back();
    const Node&A=T[ia]; const Node&B=T[ib];
    if(ia==ib){ if(A.left<0) continue;
      st.push_back({A.left,A.left}); st.push_back({A.right,A.right}); st.push_back({A.left,A.right}); continue; }
    if(sep(ia,ib)){ if(getenv("NOCERT")){ ++pairs; mass_tot+=(long long)(A.end-A.begin)*(B.end-B.begin); continue; }
      ++pairs; const long long m=(long long)(A.end-A.begin)*(B.end-B.begin); mass_tot+=m;
      long long cred[3]={0,0,0}; unsigned mask=7u;
      std::vector<std::pair<int,unsigned>> ds{{0,7u}};
      while(!ds.empty()){
        auto[ic,dm]=ds.back(); ds.pop_back(); const Node&C=T[ic];
        const bool ovl=!(C.end<=A.begin||C.begin>=A.end)||!(C.end<=B.begin||C.begin>=B.end);
        if(ovl){ if(C.left>=0){ds.push_back({C.left,dm});ds.push_back({C.right,dm});} continue; }
        ++evals;
        const i64 L=Lambda(A.box,B.box,C.box);
        unsigned got=0;
        if(L>0){ const i128 LL=(i128)L*L;
          const i128 EX=(i128)maxsq(C.box,A.box)*(i128)maxsq(B.box,C.box);
          got|=1u; if(4*LL>EX) got|=2u; if(3*LL>EX) got|=4u; }
        const unsigned take=got&dm;
        if(take){ ++cnodes; const long long k=C.end-C.begin;
          for(int q=0;q<3;++q) if(take&(1u<<q)) cred[q]+=k; }
        bool done=true; for(int q=0;q<3;++q) if(cred[q]<need[q]) done=false;
        if(done) break;
        const unsigned rest=dm&~take;
        if(rest&&C.left>=0){ ds.push_back({C.left,rest}); ds.push_back({C.right,rest}); } }
      for(int q=0;q<3;++q) if(cred[q]>=need[q]) closed[q]+=m;
      continue; }
    if(A.rad>=B.rad){ if(A.left<0){ st.push_back({ib,ia}); continue; }
      st.push_back({A.left,ib}); st.push_back({A.right,ib}); }
    else { if(B.left<0){ st.push_back({ib,ia}); continue; }
      st.push_back({ia,B.left}); st.push_back({ia,B.right}); } }
  const long long tot=(long long)n*(n-1)/2;
  printf("n=%6d s=%.1f | paires WSPD=%8lld  paires/n=%6.2f  masse=%10lld/%lld | q2 %6.2f%% q3 %6.2f%% q4 %6.2f%% | eval=%lld eval/n=%.1f\n",
    n,s,pairs,(double)pairs/n,mass_tot,tot,100.0*closed[0]/tot,100.0*closed[1]/tot,100.0*closed[2]/tot,evals,(double)evals/n);
  return 0;
}
