cmake_minimum_required(VERSION 3.20)
project(mhgp9_q34_cuda_waves LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MHGP9_CUDA_WAVES_ENABLE_CUDA "Compile isolated CUDA runner" OFF)
option(MHGP9_AUDIT_SANITIZE "ASan/UBSan host qualification" OFF)
get_filename_component(DEFAULT_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(MHGP9_SOURCE_ROOT "${DEFAULT_SOURCE_ROOT}" CACHE PATH "v9 source snapshot")
set(MHGP9_GEN_LIBRARY "" CACHE FILEPATH "Optional pinned local generator archive")
if(NOT EXISTS "${MHGP9_SOURCE_ROOT}/src/gen/wspd/front.hpp")
  message(FATAL_ERROR "v9 source snapshot required")
endif()
find_package(Threads REQUIRED)
if(MHGP9_GEN_LIBRARY)
  if(NOT EXISTS "${MHGP9_GEN_LIBRARY}")
    message(FATAL_ERROR "Requested generator archive missing")
  endif()
  set(GEN_TARGET "${MHGP9_GEN_LIBRARY}")
else()
  # Explicit source-list port of the engine mhgp9_gen target at 70168cc3b.
  set(GEN_SOURCES
    pipeline/local_credits.cpp pipeline/axis_q2.cpp pipeline/q2_census.cpp
    pipeline/q2_census_parallel.cpp pipeline/prepared_cloud.cpp wspd/front.cpp
    lanes/q4_family.cpp lanes/exact_ball.cpp lanes/q34_seed.cpp lanes/edge_cover.cpp
    lanes/q34_cover.cpp lanes/family_certificate.cpp lanes/q34_pruning.cpp
    lanes/q34_collective.cpp lanes/q4_positive_domain.cpp lanes/q4_center_map.cpp
    lanes/q4_local_partition.cpp lanes/q4_local.cpp lanes/q4_shallow_set.cpp
    lanes/q4_shallow.cpp lanes/q4_window.cpp pipeline/wspd_q34.cpp
    lanes/q34_witness_search.cpp lanes/q3_ball_census.cpp lanes/q34_dead_lanes.cpp)
  list(TRANSFORM GEN_SOURCES PREPEND "${MHGP9_SOURCE_ROOT}/src/gen/")
  add_library(cuda_waves_gen STATIC ${GEN_SOURCES})
  target_include_directories(cuda_waves_gen PUBLIC "${MHGP9_SOURCE_ROOT}/src/gen")
  target_compile_options(cuda_waves_gen PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(cuda_waves_gen PUBLIC Threads::Threads)
  set(GEN_TARGET cuda_waves_gen)
endif()
add_executable(mhgp9_q34_cuda_waves probe.cpp runner.cpp)
target_include_directories(mhgp9_q34_cuda_waves PRIVATE "${MHGP9_SOURCE_ROOT}/src" "${MHGP9_SOURCE_ROOT}/src/gen" "${MHGP9_SOURCE_ROOT}/tests/gen")
target_compile_options(mhgp9_q34_cuda_waves PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>)
target_link_libraries(mhgp9_q34_cuda_waves PRIVATE ${GEN_TARGET} Threads::Threads)
if(MHGP9_CUDA_WAVES_ENABLE_CUDA)
  if(MHGP9_AUDIT_SANITIZE)
    message(FATAL_ERROR "Host sanitizers and device qualification are separate")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(mhgp9_q34_cuda_waves PRIVATE runner.cu)
  set_target_properties(mhgp9_q34_cuda_waves PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES 120)
  target_compile_options(mhgp9_q34_cuda_waves PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-Wall,-Wextra>)
  target_link_libraries(mhgp9_q34_cuda_waves PRIVATE CUDA::cudart_static)
else()
  target_sources(mhgp9_q34_cuda_waves PRIVATE runner_stub.cpp)
endif()
if(MHGP9_AUDIT_SANITIZE)
  target_compile_options(mhgp9_q34_cuda_waves PRIVATE -O1 -g1 -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mhgp9_q34_cuda_waves PRIVATE -fsanitize=address,undefined)
  if(TARGET cuda_waves_gen)
    target_compile_options(cuda_waves_gen PRIVATE -O1 -g1 -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()
