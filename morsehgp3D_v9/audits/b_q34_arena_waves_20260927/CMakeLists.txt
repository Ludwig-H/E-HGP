cmake_minimum_required(VERSION 3.20)
project(mhgp9_q34_arena_waves LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT EXISTS "${MHGP9_SOURCE_ROOT}/src/gen/wspd/front.hpp" OR NOT EXISTS "${MHGP9_GEN_LIBRARY}")
  message(FATAL_ERROR "Pinned source and generator archive required")
endif()
find_package(Threads REQUIRED)
add_executable(mhgp9_q34_arena_waves probe.cpp)
target_include_directories(mhgp9_q34_arena_waves PRIVATE "${MHGP9_SOURCE_ROOT}/src" "${MHGP9_SOURCE_ROOT}/src/gen" "${MHGP9_SOURCE_ROOT}/tests/gen")
target_compile_options(mhgp9_q34_arena_waves PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(mhgp9_q34_arena_waves PRIVATE "${MHGP9_GEN_LIBRARY}" Threads::Threads)
