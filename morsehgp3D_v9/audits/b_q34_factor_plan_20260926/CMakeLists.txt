cmake_minimum_required(VERSION 3.20)
project(mhgp9_audit_q34_factor_plan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT MHGP9_SOURCE_ROOT OR NOT EXISTS "${MHGP9_SOURCE_ROOT}/src/gen/wspd/front.hpp")
  message(FATAL_ERROR "MHGP9_SOURCE_ROOT must name the explicitly pinned v9 source directory")
endif()
if(NOT MHGP9_GEN_LIBRARY OR NOT EXISTS "${MHGP9_GEN_LIBRARY}")
  message(FATAL_ERROR "MHGP9_GEN_LIBRARY must name the explicitly pinned, read-only libmhgp9_gen.a")
endif()
find_package(Threads REQUIRED)
add_executable(mhgp9_audit_q34_factor_plan probe.cpp)
target_include_directories(mhgp9_audit_q34_factor_plan PRIVATE
  "${MHGP9_SOURCE_ROOT}/src" "${MHGP9_SOURCE_ROOT}/src/gen" "${MHGP9_SOURCE_ROOT}/tests/gen")
target_compile_options(mhgp9_audit_q34_factor_plan PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(mhgp9_audit_q34_factor_plan PRIVATE "${MHGP9_GEN_LIBRARY}" Threads::Threads)
option(MHGP9_AUDIT_SANITIZE "Instrument this audit translation unit (link a matching instrumented gen archive)" OFF)
if(MHGP9_AUDIT_SANITIZE)
  target_compile_options(mhgp9_audit_q34_factor_plan PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mhgp9_audit_q34_factor_plan PRIVATE -fsanitize=address,undefined)
endif()
