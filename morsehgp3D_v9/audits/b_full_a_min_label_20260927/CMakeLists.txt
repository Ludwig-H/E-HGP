cmake_minimum_required(VERSION 3.20)
project(mhgp9_full_a_min_label LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MHGP9_MIN_LABEL_SANITIZE "ASan UBSan host gate" OFF)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_path(MHGP9_BOOST_INCLUDE_DIR boost/multiprecision/cpp_int.hpp HINTS "${BOOST_ROOT}/include" "$ENV{BOOST_ROOT}/include")
if(NOT MHGP9_BOOST_INCLUDE_DIR)
  message(FATAL_ERROR "Pinned rational fixture oracle requires Boost headers")
endif()
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../b_full_a_events_20260927/events.hpp" PORT_HASH)
if(NOT PORT_HASH STREQUAL "8c1280e9082d20e69cc1c47240160c336cb54f30d88594d34722db944d6de94f")
  message(FATAL_ERROR "Explicit event port source changed")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../b_full_a_manifest_20260927/instrument.py" --check RESULT_VARIABLE INSTRUMENT_STATUS)
if(NOT INSTRUMENT_STATUS EQUAL 0)
  message(FATAL_ERROR "Generated observation Builder does not match its pinned native source")
endif()
add_executable(mhgp9_full_a_min_label probe.cpp)
target_include_directories(mhgp9_full_a_min_label PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
target_include_directories(mhgp9_full_a_min_label SYSTEM PRIVATE "${MHGP9_BOOST_INCLUDE_DIR}")
target_compile_options(mhgp9_full_a_min_label PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(mhgp9_full_a_min_label PRIVATE Threads::Threads)
if(MHGP9_MIN_LABEL_SANITIZE)
  target_compile_options(mhgp9_full_a_min_label PRIVATE -O1 -g1 -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(mhgp9_full_a_min_label PRIVATE -fsanitize=address,undefined)
endif()
