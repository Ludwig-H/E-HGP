cmake_minimum_required(VERSION 3.20)
project(mhgp9_continuation_origin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(mhgp9_continuation_origin gate.cpp)
target_include_directories(mhgp9_continuation_origin PRIVATE "${MHGP9_SOURCE_ROOT}/src"
  "${MHGP9_SOURCE_ROOT}/src/gen" "${MHGP9_SOURCE_ROOT}/tests/gen")
target_link_libraries(mhgp9_continuation_origin PRIVATE "${MHGP9_GEN_LIBRARY}" Threads::Threads)
target_compile_options(mhgp9_continuation_origin PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MHGP9_AUDIT_SANITIZE)
  target_compile_options(mhgp9_continuation_origin PRIVATE -O1 -g1 -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(mhgp9_continuation_origin PRIVATE -fsanitize=address,undefined)
endif()
