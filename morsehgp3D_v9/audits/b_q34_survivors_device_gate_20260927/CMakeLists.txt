cmake_minimum_required(VERSION 3.20)
project(mhgp9_survivors_device_gate LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
get_filename_component(V9_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(mhgp9_survivors_device_gate device_gate.cu)
target_include_directories(mhgp9_survivors_device_gate PRIVATE "${V9_ROOT}/src")
set_target_properties(mhgp9_survivors_device_gate PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES 120)
target_compile_options(mhgp9_survivors_device_gate PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-Wall,-Wextra>)
target_link_libraries(mhgp9_survivors_device_gate PRIVATE CUDA::cudart_static)
