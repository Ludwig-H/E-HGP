cmake_minimum_required(VERSION 3.20)
project(mhgp9_full_a_manifest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MHGP9_AUDIT_SANITIZE "ASan UBSan host gate" OFF)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_path(MHGP9_BOOST_INCLUDE_DIR boost/multiprecision/cpp_int.hpp HINTS "${BOOST_ROOT}/include" "$ENV{BOOST_ROOT}/include")
if(NOT MHGP9_BOOST_INCLUDE_DIR)
  message(FATAL_ERROR "Pinned rational fixture oracle requires Boost headers")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/instrument.py" --check RESULT_VARIABLE INSTRUMENT_STATUS)
if(NOT INSTRUMENT_STATUS EQUAL 0)
  message(FATAL_ERROR "Generated Builder audit copy does not match pinned source")
endif()
add_executable(mhgp9_full_a_manifest probe.cpp)
target_compile_definitions(mhgp9_full_a_manifest PRIVATE MHGP9_TESTING)
target_include_directories(mhgp9_full_a_manifest PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
target_include_directories(mhgp9_full_a_manifest SYSTEM PRIVATE "${MHGP9_BOOST_INCLUDE_DIR}")
target_compile_options(mhgp9_full_a_manifest PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(mhgp9_full_a_manifest PRIVATE Threads::Threads)
if(MHGP9_AUDIT_SANITIZE)
  target_compile_options(mhgp9_full_a_manifest PRIVATE -O1 -g1 -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(mhgp9_full_a_manifest PRIVATE -fsanitize=address,undefined)
endif()
