"""Prepared fresh E gates: full 324, targeted Release/ASAN-UBSAN 33."""
from __future__ import annotations

import hashlib
import json
import os
from pathlib import Path
import re
import shlex
import signal
import stat
import time
import types
import xml.etree.ElementTree as ET

BASE = Path(__file__).resolve().parent
ROOT = BASE.parents[1]
SOURCE = ROOT / "morsehgp3D_v7"
E_BUILD = ROOT / "build/v7_next_q2_qualification"
E_RECEIPT = ROOT / "build/v7_next_q2_build_20260905/build_D.json"
NEW_NAME = "mhgp7_meb_lazy_q2_reject_shell"
AUTHORITIES = {
    "build/v7_next_pair_20260905/runner.py": "ae95c0cb65a1564edb2d45ae83671e204d70994d00356aea87aba147ed8ad3ba",
    "build/v7_meb_full_release/run_full_release.py": "1ad9089c4542a0d6539376c0c67a723d204417b755a41c7f4953e7c70ee61a9f",
    "build/v7_meb_full_release/expected_test_names.json": "0c85da58fdc386a00effbc82ebd85ccf4f71ac9e9f2ee329aa99ae3270154223",
    "build/v7_meb_integrated_tests/run_targeted.py": "24e9801fc03ae3c049496d0cd87567ff0edc769b94dd1ba441dc8b47f4cec3c3",
    "morsehgp3D_v7/receipts/release_20260904/run_release.py": "582922e64fcd2927be3e63c7156fefa839fd1f8c055f92a75c97514cad7a0ee5",
}
ASAN = "detect_leaks=1:halt_on_error=1"
UBSAN = "halt_on_error=1:print_stacktrace=1"


def require(condition: bool, message: str) -> None:
    if not condition:
        raise RuntimeError(message)


def load(name: str, key: str):
    path = ROOT / key
    data = path.read_bytes()
    require(hashlib.sha256(data).hexdigest() == AUTHORITIES[key], "authority pin changed: " + key)
    module = types.ModuleType(name)
    module.__file__ = str(path)
    exec(compile(data, str(path), "exec"), module.__dict__)
    return module


N = load("next_pair_authority", "build/v7_next_pair_20260905/runner.py")
F = load("full_D_authority", "build/v7_meb_full_release/run_full_release.py")
T = load("targeted_D_authority", "build/v7_meb_integrated_tests/run_targeted.py")
H = F.H


def names(mode: str) -> list[str]:
    require(mode in ("full", "release", "sanitized"), "unknown qualification mode")
    old = F.expected_names() if mode == "full" else T.NAMES
    require(NEW_NAME not in old and len(old) == (323 if mode == "full" else 32), "historical inventory changed")
    expected = sorted([*old, NEW_NAME])
    if mode == "full":
        require(json.loads((BASE / "expected_full_names.json").read_text()) == expected, "324 inventory is not exact historical plus q2")
    require(len(set(expected)) == len(expected), "duplicate expected name")
    return expected


def paths(mode: str) -> tuple[Path, Path]:
    return (E_BUILD if mode == "full" else BASE / mode, BASE / (mode + "_receipts"))


def commands(mode: str) -> list[tuple[str, list[str], int]]:
    build, out = paths(mode)
    expected = names(mode)
    selector = ["-L", "^gate$"] if mode == "full" else ["-R", "^(" + "|".join(re.escape(name) for name in expected) + ")$"]
    flags = (["-DCMAKE_CXX_FLAGS_RELEASE=-O1 -g -DNDEBUG -fsanitize=address,undefined -fno-omit-frame-pointer",
              "-DCMAKE_EXE_LINKER_FLAGS=-fsanitize=address,undefined"] if mode == "sanitized" else [])
    return [
        ("configure", ["cmake", "-S", str(SOURCE), "-B", str(build), "-DCMAKE_BUILD_TYPE=Release",
                       "-DMHGP7_ENABLE_CUDA=OFF", "-DCMAKE_EXPORT_COMPILE_COMMANDS=ON", *flags], 120),
        ("build_incremental" if mode == "full" else "build_targets",
         ["cmake", "--build", str(build), "--parallel", "2", *([] if mode == "full" else ["--target", *T.TARGETS])],
         3600 if mode == "full" else 1200),
        ("inventory", ["ctest", "--test-dir", str(build), "--show-only=json-v1", *selector], 60),
        ("ctest", ["ctest", "--test-dir", str(build), "--output-on-failure", "--no-tests=error", *selector,
                   "--parallel", "2", "--output-junit", str(out / "ctest.junit.xml")], 7200 if mode == "full" else 1200),
    ]


def environment(mode: str) -> dict:
    require(not any(value for key, value in os.environ.items() if key.startswith("LD_")),
            "nonempty LD_* is forbidden; values are not logged")
    values = {key: os.environ.get(key, "") for key in ("ASAN_OPTIONS", "UBSAN_OPTIONS", "LSAN_OPTIONS")}
    if mode == "sanitized":
        require(values == {"ASAN_OPTIONS": ASAN, "UBSAN_OPTIONS": UBSAN, "LSAN_OPTIONS": ""},
                "sanitized mode requires exact ASAN/UBSAN options and no LSAN override; never disable leak detection")
    return {"mode": mode, "sanitizer_options": values if mode == "sanitized" else "not_required_for_uninstrumented_Release",
            "locale": {"LC_ALL": "C", "LANG": "C"}, "PYTHONDONTWRITEBYTECODE": "1"}


def candidate_binding(expected_e: str, expected_receipt: str) -> dict:
    require(bool(re.fullmatch("[0-9a-f]{64}", expected_e)) and bool(re.fullmatch("[0-9a-f]{64}", expected_receipt)),
            "explicit E binary and build receipt SHA256 pins are required")
    require(H.digest(E_RECEIPT) == expected_receipt, "E build receipt does not match supplied pin")
    binding = N.R.candidate_binding(E_BUILD / "mhgp7", SOURCE, E_RECEIPT)
    require(binding["receipt_sha256"] == expected_receipt and binding["record"]["binary"]["sha256"] == expected_e,
            "E build/CLI pin mismatch")
    return binding


def frozen_inputs(mode: str, protected: dict, expected_e: str, expected_receipt: str) -> dict:
    env = environment(mode)
    for name, pin in AUTHORITIES.items():
        require(H.digest(ROOT / name) == pin, "authority drift: " + name)
    binding = candidate_binding(expected_e, expected_receipt)
    historical = N.baseline_binding()
    require(historical["protected_binaries"] == protected, "historical protected map changed")
    files = [Path(__file__), BASE / "expected_full_names.json", E_RECEIPT,
             *[ROOT / name for name in AUTHORITIES]]
    files += [Path(binding["record"][key]["path"]) for key in ("cmake_cache", "compile_database")]
    return {"files": {str(path): H.digest(path) for path in files}, "candidate_binding": binding,
            "protected_historical": {path: H.digest(Path(path)) for path in protected},
            "sources": H.source_inventory(), "environment": env}


def compile_binding(build: Path, mode: str, expected_e: str) -> dict:
    cache = (build / "CMakeCache.txt").read_text().splitlines()
    require(all(line in cache for line in ("CMAKE_BUILD_TYPE:STRING=Release", "MHGP7_ENABLE_CUDA:BOOL=OFF",
            "CMAKE_HOME_DIRECTORY:INTERNAL=" + str(SOURCE), "CMAKE_CXX_FLAGS:STRING=")), "configured source/mode mismatch")
    release_flags = "-O1 -g -DNDEBUG -fsanitize=address,undefined -fno-omit-frame-pointer" if mode == "sanitized" else "-O3 -DNDEBUG"
    linker_flags = "-fsanitize=address,undefined" if mode == "sanitized" else ""
    require("CMAKE_CXX_FLAGS_RELEASE:STRING=" + release_flags in cache and
            "CMAKE_EXE_LINKER_FLAGS:STRING=" + linker_flags in cache and
            "CMAKE_EXE_LINKER_FLAGS_RELEASE:STRING=" in cache, "unreviewed cached compile/link flags")
    records = json.loads((build / "compile_commands.json").read_text())
    targets = ["mhgp7"] if mode == "full" else T.TARGETS
    chosen = []
    for target in targets:
        matches = [row for row in records if "CMakeFiles/" + target + ".dir/" in row.get("command", "")]
        require(len(matches) == 1, "compile target missing or repeated: " + target)
        row = matches[0]
        argv = shlex.split(row["command"])
        require(Path(argv[0]).name in ("c++", "g++", "clang++"), "compiler wrapper not reviewed")
        require(Path(row["directory"]).resolve() == build.resolve() and
                Path(row["file"]).resolve().is_relative_to(SOURCE), "compile source/build mismatch")
        standard = {"-DNDEBUG", "-std=c++20", "-Wall", "-Wextra", "-Wpedantic", "-Werror"}
        allowed = standard | ({"-O1", "-g", "-fsanitize=address,undefined", "-fno-omit-frame-pointer"}
                              if mode == "sanitized" else {"-O3"})
        if target != "mhgp7":
            allowed.add("-DMHGP7_TESTING=1")
        if target == "mhgp7_meb_lazy_gate":
            allowed.add("-I" + str(SOURCE))
        require(all(argv.count(flag) == 1 for flag in allowed), "required flags absent/repeated")
        require(argv.count("-o") == 1 and argv.count("-c") == 1, "compile options malformed")
        oi, ci = argv.index("-o"), argv.index("-c")
        require(oi + 1 < len(argv) and ci + 1 < len(argv), "compile argument missing")
        require(argv[oi + 1].startswith("CMakeFiles/" + target + ".dir/") and
                (build / argv[oi + 1]).resolve().is_relative_to(build), "object target mismatch")
        require(Path(argv[ci + 1]).resolve() == Path(row["file"]).resolve(), "compile input mismatch")
        require(all(i in (0, oi, oi + 1, ci, ci + 1) or arg in allowed for i, arg in enumerate(argv)),
                "unreviewed compilation flag")
        chosen.append(row)
    require(os.access(build / "mhgp7", os.X_OK), "qualified CLI is not executable")
    if mode != "sanitized":
        require(H.digest(build / "mhgp7") == expected_e, "Release qualification CLI differs from pinned E")
    return {"commands": chosen, "configuration": {str(path): H.digest(path) for path in
            (build / "CMakeCache.txt", build / "CTestTestfile.cmake", build / "compile_commands.json")},
            "binary_role": "independently_built_instrumented_E_not_same_bytes_as_Release" if mode == "sanitized" else "pinned_E_Release"}


def read_log(path: Path) -> tuple[bytes, dict]:
    def fields(value):
        return {"device": value.st_dev, "inode": value.st_ino, "bytes": value.st_size,
                "mtime_ns": value.st_mtime_ns, "ctime_ns": value.st_ctime_ns}
    before = path.lstat()
    require(stat.S_ISREG(before.st_mode), "LastTest must be a regular file, not a symlink")
    with path.open("rb") as stream:
        opened = os.fstat(stream.fileno())
        require(fields(before) == fields(opened), "LastTest changed at open")
        data = stream.read()
        require(fields(opened) == fields(os.fstat(stream.fileno())), "LastTest changed while reading")
    require(fields(before) == fields(path.lstat()) and len(data) == before.st_size,
            "LastTest changed after reading")
    return data, {"path": str(path), "sha256": hashlib.sha256(data).hexdigest(), **fields(before)}


def copy_log(path: Path, destination: Path) -> tuple[bytes, dict, dict]:
    data, source = read_log(path)
    with destination.open("xb") as stream:
        stream.write(data)
        stream.flush()
        os.fsync(stream.fileno())
    copied, archive = read_log(destination)
    require(copied == data and archive["sha256"] == source["sha256"], "LastTest archive copy altered")
    reread, after = read_log(path)
    require(after == source and reread == data, "LastTest source changed during archive copy")
    return data, source, archive


def previous_log(build: Path, out: Path) -> dict:
    path = build / "Testing/Temporary/LastTest.log"
    if not path.exists() and not path.is_symlink():
        return {"present": False, "source": None, "archive": None}
    _data, source, archive = copy_log(path, out / "LastTest.preexisting.stdout")
    return {"present": True, "source": source, "archive": archive}


def judge_log(data: bytes, expected: list[str]) -> dict:
    text = data.decode("utf-8", errors="strict")
    separator = "-" * 58
    start = re.match(r"Start testing: [^\n]+\n" + separator + r"\n", text)
    require(start is not None, "LastTest start header missing")
    block = re.compile(
        r'([1-9][0-9]*)/([1-9][0-9]*) Testing: (mhgp7_[^\s]+)\n'
        r'\1/\2 Test: \3\nCommand: [^\n]+\nDirectory: [^\n]+\n'
        r'"\3" start time: [^\n]+\nOutput:\n' + separator + r'\n[\s\S]*?'
        r'^<end of output>\nTest time =\s*[0-9]+(?:\.[0-9]+)? sec\n' + separator + r'\n'
        r'Test Passed\.\n"\3" end time: [^\n]+\n'
        r'"\3" time elapsed: [0-9]{2}:[0-9]{2}:[0-9]{2}\n' + separator + r'\n\n', re.M)
    position, actual, indices, totals = start.end(), [], [], []
    while not text.startswith("End testing: ", position):
        match = block.match(text, position)
        require(match is not None, "LastTest test block truncated, failed, or malformed")
        index, total, name = int(match[1]), int(match[2]), match[3]
        require(index <= total, "LastTest test index exceeds inventory")
        actual.append(name)
        indices.append(index)
        totals.append(total)
        position = match.end()
    require(re.fullmatch(r"End testing: [^\n]+\n\n(?:[^\s=]+ = +[0-9]+(?:\.[0-9]+)? sec\*proc\n\n)*",
                         text[position:]) is not None, "LastTest terminal footer missing or malformed")
    require(len(actual) == len(expected) and len(set(actual)) == len(expected) and sorted(actual) == expected,
            "LastTest names missing, duplicate, or unexpected")
    require(len(set(indices)) == len(expected) and len(set(totals)) == 1, "LastTest test indices repeated or inconsistent")
    return {"tests": len(actual), "names": sorted(actual), "terminal_footer": True, "all_blocks_passed": True}


def archive_current_log(build: Path, out: Path, expected: list[str], prior: dict, started_ns: int | None) -> dict:
    path = build / "Testing/Temporary/LastTest.log"
    # Preserve bytes before judging, also for stale, truncated and failed logs.
    data, source, archive = copy_log(path, out / "LastTest.stdout")
    require(started_ns is not None, "LastTest cannot qualify without the owned CTest start boundary")
    observed_ns = time.time_ns()
    require(started_ns <= source["mtime_ns"] <= observed_ns and started_ns <= source["ctime_ns"] <= observed_ns,
            "LastTest timestamp precedes owned CTest or lies in the future")
    if prior["present"]:
        require(source["sha256"] != prior["source"]["sha256"], "LastTest content is unchanged from the preexisting log")
    judged = judge_log(data, expected)
    # Recheck both source and archive after judgment, before a passed verdict.
    final_data, final_source = read_log(path)
    final_copy, final_archive = read_log(out / "LastTest.stdout")
    require(final_source == source and final_data == data, "LastTest source changed after judgment")
    require(final_archive == archive and final_copy == data, "LastTest archive changed after judgment")
    return {"source": source, "archive": archive, "judge": judged, "fresh": True,
            "owned_ctest_not_before_ns": started_ns, "observed_ns": observed_ns}


def terminal_pass(results: dict, labels: set[str], failure: str | None, final_errors: list,
                  before, after, tested, binaries_after, binding, binding_after, inventory, junit, last_log) -> bool:
    return (failure is None and not final_errors and set(results) == labels and
            all(record.get("exit_code") == 0 and record.get("status") == "completed" for record in results.values()) and
            before == after and tested is not None and tested == binaries_after and binding is not None and
            binding == binding_after and inventory is not None and junit is not None and last_log is not None)


def execute(mode: str, expected_e: str, expected_receipt: str) -> int:
    environment(mode)
    build, out = paths(mode)
    require(not out.exists(), "receipt already exists; use a separately reviewed fresh path")
    require(build.is_dir() if mode == "full" else not build.exists(), "full build must exist; targeted builds must be fresh")
    expected = names(mode)
    protected = N.baseline_binding()["protected_binaries"]
    before = frozen_inputs(mode, protected, expected_e, expected_receipt)
    repository_before, host_before = N.R.repository_state(), N.R.host_observation()
    require(frozen_inputs(mode, protected, expected_e, expected_receipt) == before, "preflight metadata boundary drift")
    H.BUILD, H.OUT = build, out
    out.mkdir(mode=0o700)
    H.save("inputs_before.json", before)
    H.save("expected_names.json", expected)
    prior_log = previous_log(build, out)
    H.save("LastTest.prestate.json", prior_log)
    H.save("environment.json", {"schema": "mhgp7.q2-E-qualification.v1", "mode": mode,
        "build_kind": "incremental_existing_E_Release" if mode == "full" else "fresh_isolated",
        "expected_E_sha256": expected_e, "expected_E_build_receipt_sha256": expected_receipt,
        "protected_C_C_D_historical": protected, "all_test_executions": "fresh", "gate_results_reused": False,
        "repository_before": repository_before, "host_before": host_before, "environment": environment(mode),
        "public_status": "not_claimed", "gcp": "not_used"})
    results, failure, tested, binding, inventory, junit, ctest_started_ns = {}, None, None, None, None, None, None
    try:
        for label, argv, limit in commands(mode):
            require(frozen_inputs(mode, protected, expected_e, expected_receipt) == before, "inputs drift before " + label)
            if label == "ctest":
                ctest_started_ns = time.time_ns()
                H.save("LastTest.ctest_boundary.json", {"not_before_ns": ctest_started_ns,
                       "scope": "wall_clock_boundary_immediately_before_owned_CTest"})
            results[label] = H.run(label, argv, limit)
            require(results[label]["exit_code"] == 0 and results[label]["status"] == "completed", label + " failed/censored")
            require(frozen_inputs(mode, protected, expected_e, expected_receipt) == before, "inputs drift after " + label)
            if label.startswith("build_"):
                tested, binding = H.binaries(), compile_binding(build, mode, expected_e)
                if mode != "full":
                    require(sorted(Path(row["path"]).name for row in tested) == sorted(T.TARGETS), "targeted binary inventory mismatch")
                H.save("binaries_tested.json", tested)
                H.save("compile_binding.json", binding)
            elif label == "inventory":
                inventory = json.loads((out / "inventory.stdout").read_text())
                actual = [row["name"] for row in inventory.get("tests", [])]
                require(len(actual) == len(expected) and len(set(actual)) == len(expected) and sorted(actual) == expected,
                        "actual registered names differ from full required inventory")
                require(tested == H.binaries(), "tested binary drift before CTest")
            elif label == "ctest":
                junit = F.judge_junit(ET.parse(out / "ctest.junit.xml").getroot(), expected)
    except BaseException as error:
        failure = f"{type(error).__name__}: {error}"
        print("FAILURE " + failure, flush=True)
    final_errors = []
    def observe(function):
        try:
            return function()
        except BaseException as error:
            final_errors.append(f"{type(error).__name__}: {error}")
            return None
    repository_after = observe(N.R.repository_state)
    after = observe(lambda: frozen_inputs(mode, protected, expected_e, expected_receipt))
    binaries_after = observe(H.binaries)
    binding_after = observe(lambda: compile_binding(build, mode, expected_e))
    # JUnit is judged again at the terminal boundary, never inferred from rc=0.
    junit = observe(lambda: F.judge_junit(ET.parse(out / "ctest.junit.xml").getroot(), expected))
    last_log = observe(lambda: archive_current_log(build, out, expected, prior_log, ctest_started_ns))
    passed = terminal_pass(results, {label for label, _, _ in commands(mode)}, failure, final_errors,
                           before, after, tested, binaries_after, binding, binding_after, inventory, junit, last_log)
    for name, value in (("repository_after", repository_after), ("inputs_after", after),
                        ("binaries_after", binaries_after), ("compile_binding_after", binding_after)):
        H.save(name + ".json", value)
    summary = {"status": "passed" if passed else "failed_or_invalid", "failure": failure,
        "final_errors": final_errors, "mode": mode, "required_tests": len(expected), "junit": junit,
        "last_test_log": last_log, "last_test_log_qualified": last_log is not None,
        "inputs_stable": before == after, "tested_binaries_stable": tested is not None and tested == binaries_after,
        "compile_binding_stable": binding is not None and binding == binding_after, "results": results,
        "expected_E_sha256": expected_e, "protected_C_C_D_historical": protected,
        "gate_results_reused": False, "public_status": "not_claimed", "gcp": "not_used"}
    H.save("summary.json", summary)
    H.save("receipt_manifest.json", [{"path": path.name, "sha256": H.digest(path), "size": path.stat().st_size}
           for path in sorted(out.iterdir()) if path.is_file() and path.name != "receipt_manifest.json"])
    print(json.dumps({key: value for key, value in summary.items() if key != "results"}), flush=True)
    return 0 if passed else 1


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--mode", choices=("full", "release", "sanitized"), required=True)
    parser.add_argument("--expected-e-sha256")
    parser.add_argument("--build-receipt-sha256")
    parser.add_argument("--execute", action="store_true")
    args = parser.parse_args(argv)
    if not args.execute:
        print(json.dumps({"status": "prepared_not_executed", "commands": commands(args.mode),
            "expected_tests": len(names(args.mode)), "mode": args.mode, "builds": 0, "test_engine_runs": 0,
            "expected_E_sha256": args.expected_e_sha256, "execution_requires_root_GO": True,
            "required_ASAN_OPTIONS": ASAN if args.mode == "sanitized" else None,
            "required_UBSAN_OPTIONS": UBSAN if args.mode == "sanitized" else None,
            "public_status": "not_claimed", "gcp": "not_used"}, indent=2))
        return 0
    require(args.expected_e_sha256 is not None and args.build_receipt_sha256 is not None, "two explicit E pins required at execution")
    return execute(args.mode, args.expected_e_sha256, args.build_receipt_sha256)


if __name__ == "__main__":

        signal.signal(sig, lambda signum, _frame: (_ for _ in ()).throw(InterruptedError(f"signal {signum}")))
    raise SystemExit(main())
