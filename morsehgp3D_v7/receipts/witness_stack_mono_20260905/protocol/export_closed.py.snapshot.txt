def require(value: bool, reason: str) -> None:
    if not value:
        raise RuntimeError(reason)


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def no_duplicates(pairs: list[tuple]) -> dict:
    value = {}
    for key, item in pairs:
        require(key not in value, "duplicate JSON member: " + str(key))
        value[key] = item
    return value


def json_value(data: bytes):
    return json.loads(data, object_pairs_hook=no_duplicates,
                      parse_constant=lambda value: (_ for _ in ()).throw(ValueError("nonfinite JSON " + value)))


def canonical(value):
    return json_value(json.dumps(value, sort_keys=True, allow_nan=False).encode())


def json_bytes(value) -> bytes:
    return (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + "\n").encode()


def utc(value: str) -> datetime:
    require(isinstance(value, str), "UTC timestamp missing")
    result = datetime.fromisoformat(value.replace("Z", "+00:00"))
    require(result.tzinfo is not None and result.utcoffset().total_seconds() == 0, "timestamp not UTC")
    return result


def relative(path: Path) -> str:
    return str(path.relative_to(ROOT))


class Reader:
    """Caches original bytes and catches source changes before publication."""
    def __init__(self):
        self.data: dict[Path, bytes] = {}
        self.external_pins: dict[Path, str] = {}

    def read(self, path: Path) -> bytes:
        require(path.is_absolute() and path.is_relative_to(ROOT), "read outside repository")
        require(not path.is_symlink() and stat.S_ISREG(path.stat().st_mode), "non-regular source: " + str(path))
        require(path.resolve(strict=True) == path, "source path has alias/symlink: " + str(path))
        data = path.read_bytes()
        if path in self.data:
            require(data == self.data[path], "source drift during review: " + relative(path))
        else:
            self.data[path] = data
        return data

    def json(self, path: Path):
        return json_value(self.read(path))

    def pinned(self, path: Path, expected: str) -> bytes:
        data = self.read(path)
        require(sha(data) == expected, "source pin mismatch: " + relative(path))
        return data

    def watch(self, path: Path, expected: str, digest) -> None:
        require(path.is_file() and not path.is_symlink(), "missing bound artifact: " + str(path))
        require(digest(path) == expected, "bound artifact changed: " + str(path))
        if path in self.external_pins:
            require(self.external_pins[path] == expected, "conflicting artifact authority")
        self.external_pins[path] = expected

    def terminal(self, digest) -> None:
        for path, expected in self.data.items():
            require(self.read(path) == expected, "changed original source")
        for path, expected in self.external_pins.items():
            require(digest(path) == expected, "late bound artifact drift: " + str(path))


def load_authority(reader: Reader, path: str, expected: str, name: str):
    data = reader.pinned(ROOT / path, expected)
    module = types.ModuleType(name)
    module.__file__ = str(ROOT / path)
    exec(compile(data, module.__file__, "exec"), module.__dict__)
    return module


def closed_directory(reader: Reader, directory: Path, required: set[str]) -> dict[str, bytes]:
    require(directory.is_dir() and not directory.is_symlink(), "closed receipt directory absent")
    actual = {path.name for path in directory.iterdir()}
    require(actual == required, "closed receipt inventory differs: " + relative(directory))
    # Producer writes hashes.json last. Never inspect an active run without it.
    hashes = reader.json(directory / "hashes.json")
    require(isinstance(hashes, dict) and set(hashes) == required - {"hashes.json"}, "manifest is not exact")
    result = {"hashes.json": reader.read(directory / "hashes.json")}
    for name, expected in hashes.items():
        require(PurePosixPath(name).name == name and re.fullmatch("[0-9a-f]{64}", expected) is not None,
                "manifest name/hash is malformed")
        result[name] = reader.pinned(directory / name, expected)
    return result


def verify_snapshot(reader: Reader, value: dict, N, common) -> None:
    require(value["candidate_sources"] == N.R.candidate_sources(SOURCE), "candidate source snapshot drift")
    require(value["repository_sources"] == common.source_snapshot(ROOT), "repository source snapshot drift")
    require(value["loader_injection_nonempty"] == {name: False for name in N.R.INJECTION_VARIABLES}, "loader injection snapshot")
    for name, expected in value["files"].items():
        reader.watch(Path(name), expected, N.R.digest)


def verify_build(reader: Reader, N, B, common, expected_f: str, expected_receipt: str) -> tuple[dict, dict, dict[str, bytes]]:
    raw = closed_directory(reader, BUILD_OUT, BUILD_NAMES)
    values = {name: json_value(data) for name, data in raw.items() if name.endswith(".json")}
    summary, metadata = values["summary.json"], values["metadata.json"]
    before, after, record = metadata["tracking_before"], values["tracking_after.json"], values["build_D.json"]
    require(summary["status"] == "completed" and summary["error"] is None and
            summary["binding_validated"] is True and summary["sources_helpers_C_stable"] is True and
            summary["build_receipt"] == "build_D.json", "F build is not terminal completed")
    require(summary["engine_runs"] == 0 and summary["gcp"] == "not_used" and
            summary["public_status"] == "not_claimed", "build scope changed")
    require(before == after, "F build tracking changed")
    require(metadata["build_directory_was_absent"] is True and metadata["build_directory"] == str(F_BUILD), "not the fresh F build")
    require(raw["build_D.json"] == raw["build_record.provisional.json"], "accepted record differs from validated draft")
    require(sha(raw["build_D.json"]) == expected_receipt, "explicit F receipt pin mismatch")
    require(record["binary"]["sha256"] == expected_f and record["binary"]["path"] == str(F_BUILD / "mhgp7"), "explicit F target pin mismatch")
    binding = N.R.candidate_binding(F_BUILD / "mhgp7", SOURCE, BUILD_OUT / "build_D.json")
    require(binding["record"] == record and binding["receipt_sha256"] == expected_receipt, "candidate binding mismatch")
    require(before == B.tracking(common), "build tracking does not match current frozen sources")
    require(record["sources_before"] == record["sources_after"] == before["sources"], "build source lineage mismatch")
    for name, expected in before["files"].items():
        reader.watch(Path(name), expected, N.R.digest)
    expected_commands = B.commands(F_BUILD)
    require(canonical(metadata["commands"]) == canonical(expected_commands), "build plan differs from executed protocol")
    runs = values["runs.json"]
    require(len(runs) == len(expected_commands) == 4, "build must have four command records")
    previous = utc(summary["started_utc"])
    for run, (name, argv, limit) in zip(runs, expected_commands):
        require(run["name"] == name and run["argv"] == argv and run["timeout_seconds"] == limit,
                "build command identity mismatch")
        require(run["status"] == "completed" and run["returncode"] == 0 and not run["timed_out"], "build command not completed")
        start, end = utc(run["started_utc"]), utc(run["ended_utc"])
        require(previous <= start <= end and math.isfinite(run["wall_seconds"]) and
                0 <= run["wall_seconds"] <= limit, "build chronology/cost invalid")
        previous = end
    require(previous <= utc(summary["ended_utc"]), "build terminal timestamp precedes command")
    for key in ("binary", "compile_database", "cmake_cache"):
        reader.watch(Path(record[key]["path"]), record[key]["sha256"], N.R.digest)
    return ({"status": "passed", "scope": "read_only_F_build_binding_not_new_build",
             "source_manifest_sha256": sha(raw["hashes.json"]), "F_sha256": expected_f,
             "F_build_receipt_sha256": expected_receipt, "sources_helpers_C_stable": True,
             "all_four_command_records_revalidated": True, "started_utc": summary["started_utc"],
             "ended_utc": summary["ended_utc"], "build_seconds_diagnostic": runs[-1]["wall_seconds"]}, binding, raw)


def verify_pair(reader: Reader, N, common, incidence, separation: int, baseline: dict,
                binding: dict, build_review: dict) -> tuple[dict, list[dict], dict[str, bytes]]:
    directory = PAIR_DIRS[separation]
    raw = closed_directory(reader, directory, PAIR_NAMES)
    metadata, runs, summary, after = [json_value(raw[name]) for name in
                                     ("metadata.json", "runs.json", "summary.json", "snapshot_after.json")]
    require(summary["status"] in ("paired_equal", "divergent", "pair_incomplete") and
            summary["error"] is None and summary["source_stable"] is True, "pair lacks a stable terminal decision")
    require(metadata["baseline"] == baseline and metadata["candidate"] == binding, "pair binary lineage differs")
    require(metadata["schema"] == "mhgp7-mono-E-F-pair-v1" and metadata["s"] == separation and
            metadata["identity"] == N.R.IDENTITY and metadata["smax"] == 11 and
            metadata["silent_limits"] == N.R.CAPS and metadata["run_order"] == ["E", "F"], "pair identity/caps mismatch")
    require(metadata["fresh_observations_required"] is True and metadata["historical_timing_reuse"] is False and
            metadata["cpu_affinity_requested"] == [N.R.CPU] and
            metadata["timeout_per_run_seconds"] == N.R.TIMEOUT and
            metadata["rlimit_as_gib"] == N.R.ADDRESS_LIMIT_GIB and
            metadata["payload_proxy_bytes"] == N.R.MEMORY_PROXY and
            metadata["digest"] is True and metadata["archive"] is False, "pair run envelope differs")
    require(metadata["incidence_adapted_source_sha256"] == incidence.adapted_source_sha256,
            "recorded separation classifier differs from pinned adaptation")
    require(metadata["host_before"]["host_shared"] is True and metadata["gcp"] == "not_used" and
            metadata["public_status"] == "not_claimed", "scope metadata differs")
    before = metadata["snapshot_before"]
    require(before == after, "pair terminal source/helper/binary snapshot drift")
    verify_snapshot(reader, before, N, common)
    N.binding_matches_snapshot(before, baseline, binding, F_BUILD / "mhgp7", BUILD_OUT / "build_D.json")
    require(len(runs) == 2 and [run["role"] for run in runs] == ["E", "F"], "fresh E/F observations missing")
    pair_start, pair_end = utc(summary["started_utc"]), utc(summary["ended_utc"])
    require(metadata["started_utc"] == summary["started_utc"] and utc(build_review["ended_utc"]) <= pair_start,
            "pair predates its build or metadata")
    previous, replayed = pair_start, []
    for run, role, binary in zip(runs, ("E", "F"), (N.BASELINE, F_BUILD / "mhgp7")):
        require(RUN_BASE_FIELDS <= set(run), "incomplete attempt record")
        start, end = utc(run["started_utc"]), utc(run["ended_utc"])
        require(previous <= start <= end <= pair_end, "E/F temporal order mismatch")
        require(run["separation"] == separation and run["command"] == N.command(binary, separation), "executed command identity differs")
        for field, suffix in (("stdout", ".out"), ("stderr", ".err"), ("usage", ".time")):
            require(run[field] == role + suffix, "raw file binding differs")
        expected_wrapped = ["/usr/bin/taskset", "--cpu-list", str(N.R.CPU), "/usr/bin/time", "-v", "-o",
                            str(directory / (role + ".time")), "--", *run["command"]]
        require(run["wrapped_command"] == expected_wrapped, "fresh run does not target this receipt directory")
        require(type(run["returncode"]) is int and type(run["timed_out"]) is bool and
                isinstance(run["wall_seconds"], (int, float)) and math.isfinite(run["wall_seconds"]) and
                run["wall_seconds"] >= 0, "unterminated/nonfinite run")
        replay = {key: copy.deepcopy(run[key]) for key in RUN_BASE_FIELDS}
        replay["status"] = "invalid"
        N.classify(replay, directory / replay["stdout"], directory / replay["stderr"],
                   directory / replay["usage"], separation, common, incidence)
        require(canonical(replay) == canonical(run), "raw reclassification differs from recorded fields")
        if replay["status"] == "engine_completed":
            require(run["wall_seconds"] > 0 and run["wall_seconds"] <= N.R.TIMEOUT + 30,
                    "completed process outside bounded envelope")
            usage = raw[role + ".time"].decode()
            timed = re.findall(r'^\s*Command being timed: "(.*)"$', usage, re.M)
            require(timed == [" ".join(run["command"])], "GNU time command does not bind this fresh observation")
        replayed.append(replay)
        previous = end
    decided = N.decision(replayed, True, separation)
    fields = set(summary) - {"started_utc", "ended_utc", "repository_change_policy"}
    require(canonical({key: summary[key] for key in fields}) == canonical(decided), "terminal decision differs from raw replay")
    require(summary["repository_change_policy"] == "HEAD_or_worktree_changes_recorded_source_snapshot_stability_required",
            "repository change policy differs")
    if decided["status"] != "paired_equal":
        require("times" not in decided and "ratio_E_over_F" not in decided, "incomplete/divergent pair received comparative times")
    review = {"status": "passed", "scope": "read_only_raw_reclassification_not_new_observation",
              "separation": separation, "source_manifest_sha256": sha(raw["hashes.json"]),
              "classification_and_all_record_fields_match": True, "terminal_decision_matches": True,
              "fresh_E_then_F_bound_to_distinct_current_receipt": True,
              "replayed_statuses": [run["status"] for run in replayed], "decision": decided,
              "started_utc": summary["started_utc"], "ended_utc": summary["ended_utc"],
              "repository_change": {
                  "head_before": metadata["repository_before"]["head"],
                  "head_after": json_value(raw["repository_after.json"])["head"],
                  "head_changed": metadata["repository_before"]["head"] != json_value(raw["repository_after.json"])["head"],
                  "policy": summary["repository_change_policy"],
                  "source_and_binary_snapshots_still_identical": True,
                  "commit_contents_not_inferred_or_reaudited_by_exporter": True,
              },
              "run_intervals": [{"role": run["role"], "started_utc": run["started_utc"], "ended_utc": run["ended_utc"]}
                                for run in replayed]}
    return review, replayed, raw


def cross_separations(pairs: dict[int, tuple]) -> dict:
    previous = None
    starts = set()
    for separation in SEPARATIONS:
        review, rows, _raw = pairs[separation]
        start, end = utc(review["started_utc"]), utc(review["ended_utc"])
        require(previous is None or previous <= start, "separation pairs overlap or are reordered")
        for row in rows:
            require(row["started_utc"] not in starts, "observation timestamp reused across pairs")
            starts.add(row["started_utc"])
        previous = end
    complete = all(row["status"] == "engine_completed" for _, rows, _ in pairs.values() for row in rows)
    result = {"all_three_terminal_attempts_present": True, "six_fresh_ordered_observations_bound": True,
              "all_six_engine_completed": complete,
              "compared_fields": ["digests", "cardinalities"],
              "work_counts_not_required_equal_across_separations": True,
              "slo": "not_evaluated", "statistical_gain": "not_claimed"}
    if complete:
        reference = canonical(pairs[8][1][0])
        result["equality"] = {field: all(canonical(row[field]) == reference[field]
                                  for _, rows, _ in pairs.values() for row in rows)
                              for field in ("digests", "cardinalities")}
        result["status"] = "objects_equal" if all(result["equality"].values()) else "objects_divergent"
        result["reference_digests"] = reference["digests"]
        result["reference_cardinalities"] = reference["cardinalities"]
    else:
        result["status"] = "not_comparable_incomplete"
    return result


def result_document(build_review: dict, pairs: dict[int, tuple], cross: dict) -> dict:
    observations = []
    for separation in SEPARATIONS:
        review, rows, _ = pairs[separation]
        item = {"separation": separation, "pair_status": review["decision"]["status"],
                "run_statuses": [row["status"] for row in rows]}
        if item["pair_status"] == "paired_equal":
            item["costs"] = review["decision"]["times"]
            item["observed_process_reduction_percent_one_pair"] = 100 * (1 - rows[1]["wall_seconds"] / rows[0]["wall_seconds"])
            item["equality"] = review["decision"]["equality"]
        observations.append(item)
    return {"schema": "mhgp7-witness-stack-mono-three-separations-results-v1",
            "status": "all_pairs_equal" if all(item["pair_status"] == "paired_equal" for item in observations) else "terminal_attempts_not_all_equal",
            "build": build_review, "observations": observations, "cross_separations": cross,
            "public_status": "not_claimed", "slo": "not_evaluated", "gcp": "not_used",
            "scope": "one_cold_E_then_F_pair_per_s_shared_host_candidate_horizontal_K1_to_10_no_archive",
            "no_estimate_50k_or_massive_or_GPU": True}

def execute(expected_f: str, expected_receipt: str) -> int:
    require(re.fullmatch("[0-9a-f]{64}", expected_f) is not None and
            re.fullmatch("[0-9a-f]{64}", expected_receipt) is not None, "two explicit complete lowercase pins required")
    require(not PUBLIC.exists(), "publication target already exists; never overwrite or resume it")
    reader = Reader()
    exporter_bytes = reader.read(Path(__file__).resolve())
    N = load_authority(reader, NEXT, NEXT_SHA, "F_export_next_authority")
    B = load_authority(reader, BUILDER, BUILDER_SHA, "F_export_builder_authority")
    N.R.environment_guard()
    common, incidence = N.helpers()
    for name, expected in N.R.HELPER_PINS.items():
        reader.pinned(ROOT / "morsehgp3D_v7/bench" / name, expected)
    reader.pinned(N.LEGACY, N.LEGACY_SHA)
    baseline = N.baseline_binding()
    require(expected_f != N.BASELINE_SHA, "F pin aliases historical E; review this build explicitly")
    for name, expected in N.RECEIPTS.items():
        reader.pinned(ROOT / name, expected)
    dependencies = dict(EXISTING_DEPENDENCIES)
    dependencies.update(N.RECEIPTS)
    dependencies.update({relative(N.R.RELEASE / name): expected for name, expected in N.R.RELEASE_PINS.items()})
    for name, expected in dependencies.items():
        reader.pinned(ROOT / name, expected)
    build_review, binding, build_raw = verify_build(reader, N, B, common, expected_f, expected_receipt)
    pairs = {s: verify_pair(reader, N, common, incidence, s, baseline, binding, build_review) for s in SEPARATIONS}
    cross = cross_separations(pairs)
    results = result_document(build_review, pairs, cross)
    payload: dict[str, bytes] = {}
    mappings = []

    def copy_original(source: Path, destination: str, data: bytes) -> None:
        require(destination not in payload and not PurePosixPath(destination).is_absolute() and
                ".." not in PurePosixPath(destination).parts, "unsafe or duplicate public path")
        require(reader.read(source) == data, "source changed before mapping")
        payload[destination] = data
        mappings.append({"source_path": relative(source), "public_path": destination,
                         "source_sha256": sha(data), "public_sha256": sha(data),
                         "source_bytes": len(data), "public_bytes": len(data), "transform": "none_byte_exact"})

    for name, data in build_raw.items():
        copy_original(BUILD_OUT / name, "build_F/" + ("SOURCE_HASHES.json" if name == "hashes.json" else name), data)
    for separation, (_review, _rows, raw) in pairs.items():
        for name, data in raw.items():
            copy_original(PAIR_DIRS[separation] / name,
                          f"pairs/s{separation}/" + ("SOURCE_HASHES.json" if name == "hashes.json" else name), data)
    for source, (expected, destination) in PROTOCOL_COPIES.items():
        copy_original(ROOT / source, destination, reader.pinned(ROOT / source, expected))
    copy_original(Path(__file__).resolve(), "protocol/export_closed.py.snapshot.txt", exporter_bytes)
    for name in ("selftest_mono.py", "PLAN_MONO.md", "preparation_mono.json"):
        copy_original(BASE / name, "protocol/exporter/" + name + ".txt", reader.read(BASE / name))
    generated = {
        "review.json": "read_only_replay_and_source_integrity_checks_from_original_receipts",
        "results.json": "derived_complete_results_with_no_comparative_times_for_non_equal_pairs",
        "README.md": "generated_scope_limits_and_observed_comparison_table",
        "dependencies.json": "explicit_pins_to_already_public_protocol_and_baseline_dependencies",
        "provenance.json": "origin_map_for_every_copied_file_and_descriptions_for_generated_files",
        "manifest.json": "public_integrity_inventory_excluding_itself_and_the_two_sum_lists",
        "SHA256SUMS": "receipt_local_hash_list_excluding_both_sum_lists",
        "SHA256SUMS.root": "same_hashes_root_relative_excluding_both_sum_lists",
    }
    payload["results.json"] = json_bytes(results)
    payload["README.md"] = readme(results, pairs, len(mappings))
    payload["dependencies.json"] = json_bytes({"schema": "mhgp7-witness-stack-publication-dependencies-v1",
        "existing_public_files": [{"repository_relative_path": name, "sha256": expected}
                                  for name, expected in sorted(dependencies.items())],
        "snapshots_are_inert": True, "restore_original_layout_and_verify_all_pins_before_any_future_replay": True,
        "baseline_qualification_not_reexecuted_or_transferred": True})
    payload["provenance.json"] = json_bytes({"schema": "mhgp7-witness-stack-mono-publication-v1", "mappings": mappings,
        "generated_files": generated, "copied_files": len(mappings), "byte_exact_files": len(mappings),
        "line_ending_rewrites": 0, "exporter_sha256": sha(exporter_bytes),
        "private_repository_root": str(ROOT), "no_binaries_published": True,
        "repository_changes": {str(s): pairs[s][0]["repository_change"] for s in SEPARATIONS},
        "source_manifests_describe_original_private_names": True, "public_status": "not_claimed"})
    reader.terminal(N.R.digest)
    # Repeat semantic source/binary binding checks after raw replay, before writes.
    require(N.baseline_binding() == baseline, "historical lineage drift after replay")
    require(N.R.candidate_binding(F_BUILD / "mhgp7", SOURCE, BUILD_OUT / "build_D.json") == binding,
            "F binding drift after replay")
    require(B.tracking(common) == json_value(build_raw["tracking_after.json"]), "build sources drift after replay")
    for s in SEPARATIONS:
        verify_snapshot(reader, json_value(pairs[s][2]["snapshot_after.json"]), N, common)
    review = {"schema": "mhgp7-witness-stack-three-pairs-read-only-review-v1", "status": "passed",
              "reviewed_utc": datetime.now(timezone.utc).isoformat(),
              "scope": "raw_parsing_hash_binding_review_only_no_new_build_engine_selftest_git_or_cloud",
              "build": build_review, "pairs": {str(s): pairs[s][0] for s in SEPARATIONS},
              "cross_separations": cross, "all_copied_bytes_verified": True,
              "copied_files": len(mappings), "source_files_read_and_pinned": len(reader.data),
              "additional_bound_artifacts_watched": len(reader.external_pins),
              "baseline_C_C_D_E_protected": baseline["protected_binaries"],
              "helpers": N.R.HELPER_PINS, "separation_adapter_sha256": incidence.adapted_source_sha256,
              "runner_sha256": NEXT_SHA, "builder_sha256": BUILDER_SHA, "exporter_sha256": sha(exporter_bytes),
              "engine_runs": 0, "builds": 0, "public_status": "not_claimed", "slo": "not_evaluated", "gcp": "not_used"}
    payload["review.json"] = json_bytes(review)
    payload["manifest.json"] = json_bytes({name: {"sha256": sha(data), "bytes": len(data)} for name, data in sorted(payload.items())})
    # Everything above is read-only. Only now is the create-only public tree opened.
    PUBLIC.mkdir(mode=0o755, exist_ok=False)
    for name, data in sorted(payload.items()):
        path = PUBLIC / name
        path.parent.mkdir(parents=True, exist_ok=True)
        with path.open("xb") as output:
            output.write(data)
            output.flush()
            os.fsync(output.fileno())
        require(path.read_bytes() == data, "public copy differs from verified bytes")
    reader.terminal(N.R.digest)
    require(B.tracking(common) == json_value(build_raw["tracking_after.json"]), "late source drift during copy")
    for s in SEPARATIONS:
        verify_snapshot(reader, json_value(pairs[s][2]["snapshot_after.json"]), N, common)
    # Sum lists are the final closure. Earlier failure leaves an unclosed partial tree.
    sums = {name: sha(data) for name, data in sorted(payload.items())}
    require(all(sha((PUBLIC / name).read_bytes()) == expected for name, expected in sums.items()),
            "public payload changed before final closure")
    for name, prefix in (("SHA256SUMS", ""), ("SHA256SUMS.root", PUBLIC_REL + "/")):
        data = "".join(f"{expected}  {prefix}{path}\n" for path, expected in sums.items()).encode()
        with (PUBLIC / name).open("xb") as output:
            output.write(data)
            output.flush()
            os.fsync(output.fileno())
    require({str(path.relative_to(PUBLIC)) for path in PUBLIC.rglob("*") if path.is_file()} ==
            set(payload) | {"SHA256SUMS", "SHA256SUMS.root"}, "unexpected file in completed publication")
    require(all(sha((PUBLIC / name).read_bytes()) == expected for name, expected in sums.items()),
            "public payload changed during final closure")
    print(json.dumps({"status": "published_closed", "public_path": PUBLIC_REL,
                      "copied_files": len(mappings), "sum_covered_files": len(sums),
                      "SHA256SUMS_sha256": sha((PUBLIC / "SHA256SUMS").read_bytes()),
                      "review_sha256": sha(payload["review.json"]), "result_status": results["status"],
                      "cross_s_status": cross["status"], "public_status": "not_claimed"}, sort_keys=True))
    return 0


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--expected-f-sha256")
    parser.add_argument("--expected-build-receipt-sha256")
    parser.add_argument("--execute", action="store_true")
    args = parser.parse_args(argv)
    if not args.execute:
        print(json.dumps({"status": "prepared_not_executed", "writes": False, "receipt_reads": False,
                          "build_directory": relative(BUILD_OUT),
                          "pair_directories": [relative(PAIR_DIRS[s]) for s in SEPARATIONS],
                          "public_target": PUBLIC_REL, "requires_root_GO_and_three_terminal_pairs": True,
                          "engine_runs": 0, "gcp": "not_used"}, sort_keys=True))
        return 0
    require(args.expected_f_sha256 is not None and args.expected_build_receipt_sha256 is not None,
            "two explicit F/build receipt pins are required after root GO")
    return execute(args.expected_f_sha256, args.expected_build_receipt_sha256)


if __name__ == "__main__":
    raise SystemExit(main())
