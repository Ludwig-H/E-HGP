def load_pinned(path, expected, name):
    import hashlib
    data = path.read_bytes()
    if path.is_symlink() or hashlib.sha256(data).hexdigest() != expected:
        raise RuntimeError("module_pin_mismatch: " + str(path))
    module = types.ModuleType(name)
    module.__file__ = str(path)
    exec(compile(data, str(path), "exec"), module.__dict__)
    return module

def failed_capture(runner):
    folder = V1 / "build_20260905"
    require(digest(folder / "receipt.json") == FAILED_SHA, "failed_receipt_pin_mismatch")
    receipt = read_json(folder / "receipt.json")
    require(receipt["status"] == "failed" and
            receipt["errors"] == ["RuntimeError: unexpected_command_terminal: compile"] and
            receipt["binary_sha256"] is None and receipt["build"] is None and
            receipt["sources_stable"] is True and receipt["source_before"] == receipt["source_after"],
            "v1_failure_identity_changed")
    require(len(receipt["artifacts"]) == 16 and
            {p.name for p in folder.iterdir()} == set(receipt["artifacts"]) | {"receipt.json"},
            "failed_capture_inventory")
    data = {}
    for name, row in receipt["artifacts"].items():
        require(len(safe_relative(name).parts) == 1, "nonflat_failed_artifact")
        data[name] = read_bytes(folder / name)
        require(sha(data[name]) == row["sha256"] and len(data[name]) == row["bytes"], "failed_artifact_drift")
    commands = runner.build_commands()[:2]
    require(set(receipt["commands"]) == {"compiler", "compile"}, "failed_command_inventory")
    for label, argv, expected in commands:
        result, attempt = (runner.json_bytes(data[label + suffix]) for suffix in (".result.json", ".attempt.json"))
        spawned = runner.json_bytes(data[label + ".spawned.json"])
        require(result == receipt["commands"][label] and result["argv"] == attempt["argv"] == argv and
                result["expected_rc"] == attempt["expected_rc"] == expected == 0 and
                result["returncode"] == (1 if label == "compile" else 0) and
                result["status"] == "completed" and result["error"] is None and result["timed_out"] is False and
                attempt["cwd"] == str(ROOT) and type(spawned["pid"]) is int and spawned["pid"] > 0 and
                spawned["pgid"] == spawned["pid"], "failed_command_identity_changed")
        for stream in ("stdout", "stderr"):
            require(sha(data[label + "." + stream]) == result[stream + "_sha256"], "failed_command_stream_drift")
    require(data["compile.stdout"] == b"" and bool(data["compile.stderr"]), "failed_compile_evidence_missing")
    return receipt, data


def environment_and_scope(receipt, data, runner, cpu, seconds, stage):
    require(receipt["schema"] == "mhgp7-private-dual-budget-cost-capture-v1" and
            receipt["stage"] == stage and receipt["measurement_executed"] is (stage == "measure") and
            receipt["deadline_seconds"] == seconds and receipt["gcp_used"] is False and
            receipt["public_status"] == "not_claimed" and
            receipt["scope"] == "private_local_O2_helper_plus_capture_not_CLI_or_tower",
            "cost_capture_scope_changed")
    elapsed = (datetime.fromisoformat(receipt["finished_at"]) - datetime.fromisoformat(receipt["started_at"])).total_seconds()
    require(0 <= elapsed <= seconds, "recorded_deadline_exceeded")
    before, after, restored = (runner.json_bytes(data[name]) for name in
                              ("environment.json", "environment_after.json", "affinity_restored.json"))
    require(before["cpu_affinity"] == after["cpu_affinity"] == [cpu] and
            before["hardware"] == after["hardware"] and
            before["hardware"]["selected_cpu"]["processor"] == str(cpu) and
            restored["cpu_affinity"] == before["affinity_before"] and
            before["thread_values_effective"] == {name: "1" for name in runner.THREAD_ENV} and
            set(before["forbidden_overrides"]) == set(runner.G.FORBIDDEN_ENV) and
            not any(before["forbidden_overrides"].values()), "recorded_environment_drift")
    for label in receipt["commands"]:
        attempt = runner.json_bytes(data[label + ".attempt.json"])
        require(attempt["cpu_affinity"] == [cpu] and 0 < attempt["timeout_seconds"] <= seconds,
                "recorded_command_cpu_or_deadline")

def build_plan(reviews):
    sources, rows = {}, []
    for review in reviews:
        for name, pin in review["sources"].items():
            require(name not in sources or sources[name] == pin, "conflicting_source_pins")
            sources[name] = pin
        for name, pin in review["payload"].items():
            path = Path(name)
            compressed = path in COMPRESSED
            require(not compressed or COMPRESSED[path] == pin, "compression_input_pin")
            rows.append(dict(source=name, target="runs/" + review["name"] + "/" + path.name + (".gz" if compressed else ""),
                             sha256=pin, kind="closed_capture", compressed=compressed))
    require(len(sources) == 76 and len(rows) == 56, "cost_source_or_capture_projection")
    exclusions = []
    for name, pin in sorted(sources.items()):
        path = Path(name)
        if path in EXCLUDED:
            require(pin == EXCLUDED[path] and digest(path) == pin and read_bytes(path).startswith(b"\x7fELF"), "excluded_ELF_drift")
            exclusions.append(dict(path=name, sha256=pin))
        else:
            relative = str(path.relative_to(ROOT))
            rows.append(dict(source=name, target="source_tree/" + relative + ".txt", restore_path=relative,
                             sha256=pin, kind="inert_source_snapshot", compressed=False))
    require(sum(row["kind"] == "inert_source_snapshot" for row in rows) == 73 and len(exclusions) == 3,
            "cost_source_snapshot_count")
    exclusions.append(dict(path=str(V2 / "build_20260905/cost_harness"), sha256=BINARY_SHA))
    for base, (count, expected) in PREP_SUMS.items():
        sums = base / "SHA256SUMS"
        require(digest(sums) == expected, "preparation_sums_drift")
        lines = read_bytes(sums).decode("utf-8").splitlines()
        require(len(lines) == count, "preparation_inventory_count")
        prepared = {}
        for line in lines:
            pin, name = line.split("  ", 1)
            path = base / safe_relative(name)
            require(path not in prepared and digest(path) == pin, "preparation_file_changed_or_duplicate")
            prepared[path] = pin
        prepared[sums] = expected
        for path, pin in prepared.items():
            if str(path) not in sources:
                rows.append(dict(source=str(path), target="preparation/" + base.name + "/" + str(path.relative_to(base)) + ".txt",
                                 sha256=pin, kind="closed_preparation", compressed=False))
    rows.append(dict(source=str(REVIEW), target="review/disassembly_review_root.md.txt", sha256=REVIEW_SHA,
                     kind="root_pre_measurement_review", compressed=False))
    require(len(rows) == 151 and len({r["target"] for r in rows}) == 151 and sum(r["compressed"] for r in rows) == 2,
            "cost_copy_projection_or_compression_count")
    return rows, exclusions


def encoded(data, compressed):
    if not compressed:
        return data
    output = io.BytesIO()
    with gzip.GzipFile(filename="", mode="wb", fileobj=output, compresslevel=9, mtime=0) as stream:
        stream.write(data)
    return output.getvalue()


def restored(data, compressed):
    return gzip.decompress(data) if compressed else data


def copy_artifact(row, folder):
    raw = read_bytes(Path(row["source"]))
    require(sha(raw) == row["sha256"] and not raw.startswith(b"\x7fELF"), "raw_source_drift_or_ELF")
    archive = encoded(raw, row["compressed"])
    require(restored(archive, row["compressed"]) == raw, "compression_roundtrip_failure")
    target = folder / safe_relative(row["target"])
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open("xb") as stream:
        stream.write(archive)
    require(read_bytes(target) == archive and restored(read_bytes(target), row["compressed"]) == raw and
            read_bytes(Path(row["source"])) == raw, "copy_or_raw_source_changed")
    return dict(row, raw_bytes=len(raw), raw_sha256=sha(raw), archived_bytes=len(archive), archived_sha256=sha(archive),
                transformation="gzip_level9_mtime0_filename_empty_no_LF_change" if row["compressed"] else "byte_exact_no_LF_change")


def verify_copies(folder, provenance, check_sources):
    for row in provenance:
        archive = read_bytes(folder / safe_relative(row["target"]))
        require(len(archive) == row["archived_bytes"] and sha(archive) == row["archived_sha256"], "archived_copy_drift")
        raw = restored(archive, row["compressed"])
        require(len(raw) == row["raw_bytes"] and sha(raw) == row["raw_sha256"] == row["sha256"] and
                not raw.startswith(b"\x7fELF"), "decompressed_copy_or_raw_pin_drift")
        if row["compressed"]:
            require(archive[:10] == b"\x1f\x8b\x08\x00\x00\x00\x00\x00\x02\xff" and
                    row["transformation"] == "gzip_level9_mtime0_filename_empty_no_LF_change", "gzip_header_or_transform")
        else:
            require(row["transformation"] == "byte_exact_no_LF_change", "unexpected_copy_transform")
        if check_sources:
            require(read_bytes(Path(row["source"])) == raw, "source_changed_during_publication")


def verify_archive(folder):
    require(not (folder / "EXPORT_FAILURE.json").exists(), "failed_archive_not_closed")
    manifest = read_json(folder / "manifest.json")
    require(manifest["schema"] == "mhgp7-private-cost-publication-v1" and
            manifest["status"] == "completed_export_including_historical_failed_build" and
            manifest["public_status"] == "not_claimed", "cost_archive_manifest_status")
    entries = manifest["entries"]
    names = [str(safe_relative(row["path"])) for row in entries]
    files = list(folder.rglob("*"))
    require(len(names) == len(set(names)) and not any(p.is_symlink() for p in files) and
            {str(p.relative_to(folder)) for p in files if p.is_file()} == set(names) | {"manifest.json", "SHA256SUMS"},
            "archive_inventory_mismatch")
    for row in entries:
        data = read_bytes(folder / row["path"])
        require(sha(data) == row["sha256"] and len(data) == row["bytes"], "sealed_artifact_drift")
    provenance = read_json(folder / "provenance.json")
    require(len(provenance) == manifest["copied_artifacts"], "provenance_count_mismatch")
    verify_copies(folder, provenance, False)
    expected = "".join(digest(folder / n) + "  " + n + "\n" for n in sorted(names + ["manifest.json"]))
    require(read_bytes(folder / "SHA256SUMS") == expected.encode(), "archive_sums_mismatch")
    return dict(files=len(names) + 2, copied_artifacts=len(provenance), manifest_sha256=digest(folder / "manifest.json"),
                SHA256SUMS_sha256=digest(folder / "SHA256SUMS"))


def execute(expected):
    require(not DESTINATION.exists(), "public_destination_already_exists")
    require(digest(BASE / "export.py") == expected, "exporter_pin_mismatch")
    reviews = review_inputs()
    rows, exclusions = build_plan(reviews)
    snapshots = {BASE / name: name for name in ("export.py", "selftest.py", "README.md", "preparation.json")}
    snapshots.update({HELPER: "geometry_export.py", RESTORE: "restore_sources.py"})
    require(digest(RESTORE) == RESTORE_SHA and digest(HELPER) == HELPER_SHA, "reused_helpers_drift")
    for path, name in snapshots.items():
        rows.append(dict(source=str(path), target="exporter/" + name + ".txt", sha256=digest(path),
                         kind="inert_exporter_snapshot", compressed=False))
    require(all(digest(Path(row["source"])) == row["sha256"] for row in rows), "prepublication_source_drift")
    DESTINATION.mkdir(parents=True, exist_ok=False)
    try:
        provenance = [copy_artifact(row, DESTINATION) for row in rows]
        save(DESTINATION / "provenance.json", provenance)
        save(DESTINATION / "excluded_artifacts.json", exclusions)
        save(DESTINATION / "validation.json", dict(status="historical_failed_and_completed_captures_replayed_read_only",
             reviews=reviews, engine_runs=0, public_status="not_claimed", geometry_reference=dict(
             path="../meb_dual_geometry_20260905", manifest_sha256=GEOMETRY_MANIFEST_SHA, SHA256SUMS_sha256=GEOMETRY_SUMS_SHA)))
        save(DESTINATION / "reconstruction_manifest.json", [dict(archived_path=row["target"], restored_path=row["restore_path"], sha256=row["sha256"])
             for row in rows if row["kind"] == "inert_source_snapshot"])
        with (DESTINATION / "README.md").open("x", encoding="utf-8") as stream:
            stream.write(PUBLIC_README)
        require(review_inputs() == reviews, "qualification_changed_at_export_boundary")
        verify_copies(DESTINATION, provenance, True)
        entries = [dict(path=str(p.relative_to(DESTINATION)), sha256=digest(p), bytes=len(read_bytes(p)))
                   for p in sorted(DESTINATION.rglob("*")) if p.is_file()]
        save(DESTINATION / "manifest.json", dict(schema="mhgp7-private-cost-publication-v1",
             status="completed_export_including_historical_failed_build", entries=entries, copied_artifacts=len(provenance),
             source_snapshots=73, compressed_artifacts=2, excluded_ELF=4, engine_runs=0, public_status="not_claimed"))
        with (DESTINATION / "SHA256SUMS").open("x", encoding="utf-8") as stream:
            for path in sorted(DESTINATION.rglob("*")):
                if path.is_file() and path.name != "SHA256SUMS":
                    stream.write(digest(path) + "  " + str(path.relative_to(DESTINATION)) + "\n")
        closure = verify_archive(DESTINATION)
        verify_copies(DESTINATION, provenance, True)
        print(json.dumps(dict(status="completed_export_including_historical_failed_build", destination=str(DESTINATION),
                             closure=closure, engine_runs=0)))
        return 0
    except BaseException as error:
        save(DESTINATION / "EXPORT_FAILURE.json", dict(status="failed_or_invalid", error=f"{type(error).__name__}: {error}",
             partial_artifacts_preserved=True, never_resume=True))
        raise

def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--execute", action="store_true")
    parser.add_argument("--expected-exporter-sha256")
    args = parser.parse_args(argv)
    if not args.execute:
        print(json.dumps(dict(status="prepared_not_executed", writes=False, engine_runs=0, destination=str(DESTINATION),
             input_artifacts=151, source_snapshots=73, compressed_artifacts=2, exporter_snapshots=6,
             expected_public_files=164, requires_separate_GO=True), indent=2))
        return 0
    require(isinstance(args.expected_exporter_sha256, str) and len(args.expected_exporter_sha256) == 64 and
            all(c in "0123456789abcdef" for c in args.expected_exporter_sha256), "explicit_exporter_pin_required")
    return execute(args.expected_exporter_sha256)


if __name__ == "__main__":
    raise SystemExit(main())
