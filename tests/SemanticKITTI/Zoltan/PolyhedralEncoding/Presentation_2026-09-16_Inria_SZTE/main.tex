% 4. Intuition du contact et propriété convexe, sans démonstration.
\begin{frame}{La fonction support}
\begin{columns}[c,onlytextwidth]
\begin{column}{.49\textwidth}
Pour une direction unitaire $u$ :
\[
\boxed{h_P(u)=\max_{x\in P}\ip{u}{x}.}
\]
Hauteur du premier contact avec un plan de normale $u$.\par\bigskip
Si $P$ est \textbf{convexe}, tous ses plans d'appui déterminent $P$.
\end{column}
\begin{column}{.48\textwidth}\centering
\resizebox{\linewidth}{!}{\input{figs/plan_support}}
\end{column}
\end{columns}
\source{$P$ compact non vide. Figure : coupe 2D du déplacement d'un plan en dimension 3.}
\end{frame}

% 6. Même convention H_P que dans la conversation préparatoire.
\begin{frame}{Support quadratique et distance}
\[
\underbrace{\max_{x\in P}\bigl(2\ip{b}{x}-\norm{x}^2\bigr)}_{H_P(b)}
=\norm{b}^2-d_P(b)^2.
\]
\vspace{.1cm}
\begin{columns}[c,onlytextwidth]
\begin{column}{.45\textwidth}\centering
\begin{tikzpicture}[scale=.87]
\useasboundingbox(-.45,-.3)rectangle(4.0,3.3);
\draw[inria-2024-bleu-canard,line width=1.8pt](.1,.15)--(.9,.45)--(1.3,1.4)--(1.3,2.7)--(.6,3.05);
\node[text=inria-2024-bleu-canard]at(.35,1.65){$P$};
\draw[inria-rouge!65,dashed](2.5,1.6)circle(1.2);
\fill[inria-rouge](2.5,1.6)circle(2.3pt)node[right=3pt]{$b$};
\fill[inria-rouge](1.3,1.6)circle(2.3pt);
\draw[inria-rouge,thick](1.3,1.6)--(2.5,1.6)node[midway,above=3pt]{$d_P(b)$};
\end{tikzpicture}
\end{column}
\begin{column}{.51\textwidth}
$b$ est une \textbf{sonde dans l'espace}.
\[
d_P(b)=\min_{x\in P}\norm{b-x}.
\]
Distance aux primitives entières, pas seulement aux sommets.\par\medskip
{\small $H_P$ est convexe, même si $P$ ne l'est pas.}
\end{column}
\end{columns}
\end{frame}

% 7. Trois propriétés utiles, sans preuve ni borne de discrétisation.
\begin{frame}{Les propriétés utiles, sans convexité}
\vspace{.1cm}
\begin{tabular}{@{}p{.22\textwidth}p{.73\textwidth}@{}}
\textbf{Géométrie} & $\displaystyle P=\{b:d_P(b)=0\}$ \\[.65cm]
\textbf{Stabilité} & $\displaystyle \|d_P-d_Q\|_\infty=d_H(P,Q)$ \\[.65cm]
\textbf{Union} & $\displaystyle d_{P\cup Q}=\min(d_P,d_Q)$
\end{tabular}
\vspace{.6cm}
\par\small
Pour des compacts non vides : arêtes, surfaces ouvertes ou unions de morceaux.
\source{$d_H$ : distance de Hausdorff. Les fonctions sont comparées aux mêmes positions, dans un repère commun.}
\end{frame}

% 8. Choix final de normalisation : demi-plus-grand-côté ; aucune rotation locale.
\begin{frame}{Un vecteur par polyèdre}\vspace{.2cm}
\begin{columns}[c,onlytextwidth]
\begin{column}{.33\textwidth}\centering
\resizebox{!}{2.65cm}{\input{figs/grille}}\par{\small $8^3=512$ sondes}
\end{column}
\begin{column}{.63\textwidth}
$c_v$ : centre de la boîte englobante.\par\medskip
$s_v$ : moitié du plus grand côté.\par\medskip
$\widehat P_v=(P_v-c_v)/s_v\subset[-1,1]^3$.\par\medskip
\textbf{Aucune rotation locale.}
\end{column}
\end{columns}
\vspace{.2cm}
\[
\boxed{D_v[j]=\frac{d(c_v+s_vb_j,P_v)}{s_v},\quad j=1,\ldots,512.}
\]
\begin{center}\small
Conserver aussi : centre, taille, niveau HGP, attributs LiDAR.
\end{center}
\source{Grille $8^3$ aux centres des cellules. Code approché ; cas ponctuels traités séparément. Antécédent : Prokudin \emph{et al.}, Basis Point Sets, ICCV 2019.}
\end{frame}
