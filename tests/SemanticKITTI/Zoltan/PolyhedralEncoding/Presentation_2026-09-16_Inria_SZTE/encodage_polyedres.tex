\begin{frame}{La fonction support \citb{schneider}}
\begin{columns}[c,onlytextwidth]
\begin{column}{.49\textwidth}
Pour une direction unitaire $u$ :
\[
\boxed{h_P(u)=\max_{x\in P}\ip{u}{x}.}
\]
Hauteur du premier contact avec un plan de normale $u$.\par\medskip
Si $P$ est \textbf{convexe}, tous ses plans d'appui déterminent $P$.
\[h_P=h_{\conv(P)}.\]
\end{column}

\begin{frame}{Support quadratique et distance}
\[
H_P(b)=\max_{x\in P}\bigl(2\ip{b}{x}-\norm{x}^2\bigr)
      =\norm{b}^2-d_P(b)^2.
\]

\begin{column}{.53\textwidth}
\[
d_P(b)=\min_{x\in P}\norm{b-x}.
\]

\small
Relèvement : $C_P=\conv\{(x,\norm{x}^2):x\in P\}$,
\quad $H_P(b)=h_{C_P}(2b,-1)$.
\source{$H_P$ est convexe, même si $P$ ne l'est pas.}

\begin{frame}[t]{Distances, primitives et fusions}
\[
\boxed{d_P(b)=\min_{\sigma\in\mathcal F}d(b,\sigma).}
\]
{\small Distance aux primitives \textbf{entières}, pas seulement aux sommets.\par}
\vspace{.14cm}
\begin{tabular}{@{}p{.24\textwidth}p{.72\textwidth}@{}}
\textbf{Géométrie} & $P=\{b:d_P(b)=0\}$\\[.25cm]
\textbf{Stabilité} & $\|d_P-d_Q\|_\infty=d_H(P,Q)$\\[.25cm]
\textbf{Union} & $d_{P\cup Q}=\min(d_P,d_Q)$
\end{tabular}\par

\end{frame}
