\[
H_P(b)=\max_{x\in P}\bigl(2\ip{b}{x}-\norm{x}^2\bigr)

\begin{column}{.53\textwidth}
\[
d_P(b)=\min_{x\in P}\norm{b-x}.
\]
\textbf{Calculer directement la distance en 3D.}\par\medskip
Ni signe intérieur/extérieur,\\
ni convexe 4D à construire.
\end{column}
\end{columns}
\vspace{.13cm}
\small
Relèvement : $C_P=\conv\{(x,\norm{x}^2):x\in P\}$,
\quad $H_P(b)=h_{C_P}(2b,-1)$.
\source{Les sondes $b$ choisissent des directions du support relevé, pas toute la sphère $S^3$. Relèvement de tout $P$ ; $H_P$ est convexe même si $P$ ne l'est pas.}
\end{frame}

\begin{frame}[t]{Distances, primitives et fusions}
\[
\boxed{d_P(b)=\min_{\sigma\in\mathcal F}d(b,\sigma).}
\]
{\small Distance aux primitives \textbf{entières}, pas seulement aux sommets.\par}
\vspace{.14cm}
\begin{tabular}{@{}p{.24\textwidth}p{.72\textwidth}@{}}
\textbf{Géométrie} & $P=\{b:d_P(b)=0\}$\\[.25cm]
\textbf{Stabilité} & $\|d_P-d_Q\|_\infty=d_H(P,Q)$\\[.25cm]
\textbf{Union} & $d_{P\cup Q}=\min(d_P,d_Q)$
\end{tabular}\par
\vspace{.27cm}
\small
Après une fusion : \textbf{réévaluer aux sondes du parent}.\\
Ne pas prendre le minimum des tableaux enfants.
\source{Identités pour des compacts non vides, dans un même repère. $d_H$ : distance de Hausdorff. La stabilité concerne la géométrie donnée, pas toutes les occultations.}
\end{frame}
